\section{Synthesis my design}

Nhóm em sử dụng Standard Cells là:

\begin{itemize}[label=+, leftmargin=2cm]
	\item slow\_vdd1v2\_basicCells\_lvt.lib
	\item slow\_vdd1v2\_basicCells\_lvt.v
\end{itemize}

Tiến hành sử dụng file .tcl sau để thực hiện trên Genus:

\begin{lstlisting}[style=StyleCode, language=TCL, caption={genus\_synthesis.tcl}]
	
	##################################################
	#
	# EE3213_04 (cre: Khai Pham)
	# Digital system design and Verification
	#
	##################################################
	set top_module FPU_unit
	set_db library [list ./../slow_vdd1v2_basicCells_lvt.lib ]
	#HDL files read#
	read_hdl -sv -f ./flist.f
	elaborate
	set_top_module ${top_module}
	write_hdl > ./Genus/${top_module}_elab.v
	check_design -all
	read_sdc ./constraint.sdc
	syn_generic
	write_hdl > ./Genus/${top_module}_generic.v
	syn_map
	write_hdl > ./Genus/${top_module}_tech_map.v
	remove_assigns_without_optimization -verbose
	syn_opt  
	report timing -lint
	write_sdf -edges check_edge -setuphold "split" -recrem split > ./Genus/${top_module}.sdf
	report timing -max_paths 10 > ./Genus/report/${top_module}.timing.rpt
	report hierarchy > ./Genus/report/${top_module}.hier.rpt
	report gates > ./Genus/report/${top_module}.gates.rpt
	report datapath > ./Genus/report/${top_module}.datapath.rpt
	report qor > ./Genus/report/${top_module}.qor.rpt
	report area > ./Genus/report/${top_module}.area.rpt
	report power > ./Genus/report/${top_module}.power.rpt
	gui_show
\end{lstlisting}

Với file \textbf{constraint.sdc} như sau:

\begin{lstlisting}[style=StyleCode, language=TCL, caption={File constraint.sdc}]
	set_max_delay 2.0 -from [all_inputs] -to [all_outputs]
	set_input_delay 0.1 -clock [get_clocks *] [all_inputs]
	set_output_delay 0.1 -clock [get_clocks *] [all_outputs]
	set timing_report_unconstrained true
\end{lstlisting}
